const API_BASE_URL =
  (import.meta.env.VITE_API_BASE_URL as string | undefined) ||
  "http://localhost:5000/api";

const AUTH_TOKEN_KEY = "truehire_auth_token";

// Thrown for every failed request. `message` is safe to show to the user.
export class ApiError extends Error {
  readonly status: number;
  readonly code?: string;

  constructor(message: string, status: number, code?: string) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.code = code;
  }
}

interface ErrorBody {
  message?: string;
  error?: string;
  code?: string;
}

export async function apiClient<T>(
  endpoint: string,
  options: RequestInit = {},
): Promise<T> {
  const headers = new Headers(options.headers);

  // A browser must set the multipart boundary itself, so only JSON bodies get a type.
  const isFormData = options.body instanceof FormData;
  if (!isFormData && options.body !== undefined && !headers.has("Content-Type")) {
    headers.set("Content-Type", "application/json");
  }

  const token = localStorage.getItem(AUTH_TOKEN_KEY);
  if (token) {
    headers.set("Authorization", `Bearer ${token}`);
  }

  let response: Response;
  try {
    response = await fetch(`${API_BASE_URL}${endpoint}`, { ...options, headers });
  } catch {
    throw new ApiError(
      "Cannot reach the server. Check your connection and try again.",
      0,
      "NETWORK_ERROR",
    );
  }

  // An expired or revoked token signs the user out everywhere. Login and register are exempt.
  const isLoginOrRegister =
    endpoint === "/auth/login" || endpoint === "/auth/register";
  if (response.status === 401 && token && !isLoginOrRegister) {
    localStorage.removeItem(AUTH_TOKEN_KEY);
    window.dispatchEvent(new Event("truehire:unauthorized"));
  }

  if (!response.ok) {
    const body = (await response.json().catch(() => null)) as ErrorBody | null;
    throw new ApiError(
      body?.message || body?.error || `Request failed (HTTP ${response.status})`,
      response.status,
      body?.code,
    );
  }

  return response.json() as Promise<T>;
}

export function setAuthToken(token: string): void {
  localStorage.setItem(AUTH_TOKEN_KEY, token);
}

export function getAuthToken(): string | null {
  return localStorage.getItem(AUTH_TOKEN_KEY);
}

export function clearAuthToken(): void {
  localStorage.removeItem(AUTH_TOKEN_KEY);
}

import { Request, Response, NextFunction } from 'express';
import { authService } from '../services/authService.js';
import { registerSchema, loginSchema, updateProfileSchema } from '../validators/authValidator.js';
import { AuthenticatedRequest } from '../middleware/authMiddleware.js';

export async function register(req: Request, res: Response, next: NextFunction): Promise<void> {
  try {
    const validatedInput = registerSchema.parse(req.body);
    const result = await authService.register(validatedInput);
    res.status(201).json(result);
  } catch (error: any) {
    if (error.name === 'ZodError') {
      res.status(400).json({ error: 'Validation failed', details: error.errors });
      return;
    }
    if (error.message === 'User with this email already exists') {
      res.status(409).json({ error: error.message });
      return;
    }
    next(error);
  }
}

export async function login(req: Request, res: Response, next: NextFunction): Promise<void> {
  try {
    const validatedInput = loginSchema.parse(req.body);
    const result = await authService.login(validatedInput);
    res.status(200).json(result);
  } catch (error: any) {
    if (error.name === 'ZodError') {
      res.status(400).json({ error: 'Validation failed', details: error.errors });
      return;
    }
    if (error.message === 'Invalid email or password') {
      res.status(401).json({ error: error.message });
      return;
    }
    next(error);
  }
}

export async function getMe(req: Request, res: Response, next: NextFunction): Promise<void> {
  try {
    const userId = (req as AuthenticatedRequest).user?.userId;
    if (!userId) {
      res.status(401).json({ error: 'Unauthorized' });
      return;
    }
    const profile = await authService.getProfile(userId);
    res.status(200).json(profile);
  } catch (error) {
    next(error);
  }
}

export async function updateProfile(req: Request, res: Response, next: NextFunction): Promise<void> {
  try {
    const userId = (req as AuthenticatedRequest).user?.userId;
    if (!userId) {
      res.status(401).json({ error: 'Unauthorized' });
      return;
    }
    const validatedInput = updateProfileSchema.parse(req.body);
    const updatedProfile = await authService.updateProfile(userId, validatedInput);
    res.status(200).json(updatedProfile);
  } catch (error: any) {
    if (error.name === 'ZodError') {
      res.status(400).json({ error: 'Validation failed', details: error.errors });
      return;
    }
    if (error.message === 'Email is already taken by another user') {
      res.status(409).json({ error: error.message });
      return;
    }
    next(error);
  }
}

export interface UserResponse {
  id: string;
  email: string;
  name: string;
  role: string;
  targetRole?: string | null;
  preferredLocations?: string[];
  experienceLevel?: string | null;
  createdAt: Date;
  updatedAt: Date;

export interface AuthResult {
  user: UserResponse;
  token: string;
}

export class AuthService {
  constructor(private userRepo: UserRepository = userRepository) {}

  private sanitizeUser(user: any): UserResponse {
    return {
      id: user.id,
      email: user.email,
      name: user.name,
      role: user.role,
      targetRole: user.targetRole || null,
      preferredLocations: user.preferredLocations || [],
      experienceLevel: user.experienceLevel || null,
      createdAt: user.createdAt,
      updatedAt: user.updatedAt,
    };
  }

  async register(input: RegisterInput): Promise<AuthResult> {
    const existing = await this.userRepo.findByEmail(input.email);
    if (existing) {
      throw new Error("User with this email already exists");
    }

    const hashedPassword = await hashPassword(input.password);
    const user = await this.userRepo.create({
      email: input.email,
      passwordHash: hashedPassword,
      name: input.name,
    });

    const token = signToken({
      userId: user.id,
      email: user.email,
      role: user.role,
    });

    return {
      user: this.sanitizeUser(user),
      token,
    };
  }

  async login(input: LoginInput): Promise<AuthResult> {
    const user = await this.userRepo.findByEmail(input.email);
    if (!user) {
      throw new Error("Invalid email or password");
    }

    const isMatch = await comparePassword(input.password, user.passwordHash);
    if (!isMatch) {
      throw new Error("Invalid email or password");
    }

    const token = signToken({
      userId: user.id,
      email: user.email,
      role: user.role,
    });

    return {
      user: this.sanitizeUser(user),
      token,
    };
  }

  async getProfile(userId: string): Promise<UserResponse> {
    const user = await this.userRepo.findById(userId);
    if (!user) {
      throw new Error("User not found");
    }
    return this.sanitizeUser(user);
  }

  async updateProfile(
    userId: string,
    input: UpdateProfileInput,
  ): Promise<UserResponse> {
    const user = await this.userRepo.findById(userId);
    if (!user) {
      throw new Error("User not found");
    }

    const updateData: {
      name?: string;
      email?: string;
      passwordHash?: string;
      targetRole?: string;
      preferredLocations?: string[];
      experienceLevel?: string;
    } = {};

    if (input.name) {
      updateData.name = input.name;
    }

    if (input.email && input.email !== user.email) {
      const existing = await this.userRepo.findByEmail(input.email);
      if (existing) {
        throw new Error("Email is already taken by another user");
      }
      updateData.email = input.email;
    }

    if (input.password) {
      updateData.passwordHash = await hashPassword(input.password);
    }

    if (input.targetRole) {
      updateData.targetRole = input.targetRole;
    }

    if (input.preferredLocations) {
      updateData.preferredLocations = input.preferredLocations;
    }

    if (input.experienceLevel) {
      updateData.experienceLevel = input.experienceLevel;
    }

    const updatedUser = await this.userRepo.update(userId, updateData);
    return this.sanitizeUser(updatedUser);
  }
}

export const createInterviewService = async (
  userId: string,
  applicationId: string,
) => {
  const application = await prisma.application.findFirst({
    where: { id: applicationId, userId },
    include: { job: true },
  });
  if (!application || !application.job) {
    throw new AppError(404, "Application or associated job not found", "APPLICATION_NOT_FOUND");
  }

  const resume = await prisma.resume.findFirst({
    where: { userId },
    orderBy: { createdAt: "desc" },
  });
  if (!resume) {
    throw new AppError(400, "Upload a resume before starting an interview", "RESUME_REQUIRED");
  }

  // A job with no comparable skills has a null match score. It is never replaced with a number.
  const matchScore = application.matchScore ?? null;
  const topSkillGaps = application.skillGaps ?? [];

  const promptInput: InterviewPromptInput = {
    job: {
      title: application.job.title,
      company: application.job.company,
      location: application.job.location,
      description: application.job.description,
      requirements: application.job.requirements,
    },
    resume: {
      rawText: resume.rawText,
      skills: resume.skills,
      experienceYrs: resume.experienceYrs,
    },
    matchScore,
    topSkillGaps,
  };

  // All AI work happens before any write. If it fails, nothing is saved.
  // The model may echo a score. The stored value is the only one we keep.
  const generatedPlan = await generateInterviewPlan(promptInput);
  const plan = { ...generatedPlan, matchScore };
  const generatedQuestions = await generateInterviewQuestions({
    plan,
    job: application.job,
    resume,
    matchScore,
    topSkillGaps,
  });

  const title = `${application.job.title} at ${application.job.company}`;
  const interview = await createInterviewWithQuestions(
    { userId, applicationId, title, plan },
    generatedQuestions.questions,
  );

  return await prisma.interview.findUnique({
    where: { id: interview.id },
    include: { questions: true },
  });
};

export const submitAnswerService = async (
  interviewId: string,
  questionId: string,
  userId: string,
  answer: string,
) => {
  const interview = await getInterviewSessionById(interviewId, userId);
  if (!interview) {
    throw new AppError(404, "Interview session not found", "INTERVIEW_NOT_FOUND");
  }
  if (interview.status !== "IN_PROGRESS") {
    throw new AppError(
      400,
      "Answers can only be submitted while the interview is in progress",
      "INTERVIEW_NOT_IN_PROGRESS",
    );
  }

  const question = await getQuestionById(questionId, interviewId);
  if (!question) {
    throw new AppError(404, "Question not found for this interview", "QUESTION_NOT_FOUND");
  }

  // A question is final once it has feedback. Only unevaluated questions can take an answer.
  if (question.feedback !== null) {
    throw new AppError(409, "This question has already been answered", "ALREADY_ANSWERED");
  }

  // Questions are answered in order. The next one is the first without feedback.
  const nextDue = interview.questions.find((q) => q.feedback === null);
  if (!nextDue || nextDue.id !== questionId) {
    throw new AppError(
      409,
      `Answer the questions in order. Question ${nextDue?.order ?? ""} is next.`.trim(),
      "OUT_OF_ORDER",
    );
  }

  // Save first. If the evaluation below fails, the answer is still on record.
  await saveAnswer(questionId, answer);

  let evaluation: Awaited<ReturnType<typeof evaluateQuestionAnswer>>;
  try {
    evaluation = await evaluateQuestionAnswer(question.question, answer, interview.title);
  } catch {
    throw new AppError(
      503,
      "Your answer was saved, but evaluation is temporarily unavailable. Submit the same answer again to retry.",
      "EVALUATION_PENDING",
    );
  }

  const updatedQuestion = await saveQuestionAnswerAndFeedback(
    questionId,
    answer,
    JSON.stringify(evaluation),
    evaluation.score,
  );

  // Completed means no question is left without feedback, whatever order they were answered in.
  const isCompleted = (await countUnfinishedQuestions(interviewId)) === 0;
  if (isCompleted) {
    await updateInterviewStatus(interviewId, userId, "COMPLETED");
  }

  return {
    evaluation,
    updatedQuestion,
    nextQuestion: isCompleted ? null : await getFirstUnfinishedQuestion(interviewId),
    completed: isCompleted,
  };
};

export const generateFinalEvaluationService = async (
  interviewId: string,
  userId: string,
) => {
  const interview = await getInterviewSessionById(interviewId, userId);
  if (!interview) {
    throw new AppError(404, "Interview not found", "INTERVIEW_NOT_FOUND");
  }
  if (interview.status !== "COMPLETED") {
    throw new AppError(
      409,
      "Finish every question before the report is generated",
      "INTERVIEW_NOT_COMPLETE",
    );
  }
  if (await getEvaluationByInterviewId(interviewId)) {
    throw new AppError(409, "An evaluation already exists for this interview", "EVALUATION_EXISTS");
  }

  // 4. Extract role and application context
  const application = interview.application;
  const job = application?.job;
  const jobTitle = job?.title || "Software Engineer";
  const companyName = job?.company || "Target Company";
  const roleDescription = job?.description || "";
  // No match score is stored when the posting had no comparable skills. Never invent one.
  const matchScore = application?.matchScore ?? null;
  const skillGaps = application?.skillGaps || [];

  // 5. Build prompt
  const prompt = buildFinalEvaluationPrompt({
    jobTitle,
    companyName,
    roleDescription,
    matchScore,
    skillGaps,
    questions: interview.questions.map((q) => ({
      order: q.order,
      round: q.round,
      question: q.question,
      answer: q.answer,
      score: q.score,
      feedback: q.feedback,
    })),
  });

  // 6. Request evaluation from Nemotron
  const aiResponse = await analyzeResume(prompt);

  let rawEvaluation: any;

  if (typeof aiResponse === "string") {
    // If string, sanitize markdown formatting and parse
    const cleanedResponse = aiResponse
      .replace(/```json/g, "")
      .replace(/```/g, "")
      .trim();
    rawEvaluation = JSON.parse(cleanedResponse);
  } else {
    // If already parsed object, use directly
    rawEvaluation = aiResponse;
  }
  // Normalize key names if Nemotron misspells "confidence" as "confidance"
  if (rawEvaluation?.breakdown) {
    if (
      rawEvaluation.breakdown.confidance !== undefined &&
      rawEvaluation.breakdown.confidence === undefined
    ) {
      rawEvaluation.breakdown.confidence = rawEvaluation.breakdown.confidance;
      delete rawEvaluation.breakdown.confidance;
    }
  }

  const validatedEvaluation = FinalEvaluationSchema.parse(rawEvaluation);

  // 8. Persist to PostgreSQL
  // The report and its skill updates are saved together.
  const savedEvaluation = await saveEvaluationWithSkillEvidence(
    interviewId,
    userId,
    validatedEvaluation,
    skillEvidenceFromReport(validatedEvaluation),
  );

  return savedEvaluation;
};

export const matchResumeToJob = async (

// Deterministic resume-to-job matching. Every number can be traced to a skill name,
// a count of mentions in the job description, or an interview report. No model is involved.
import { normalizeSkills } from "./skillNormalizer";
import { countSkillMentions } from "./skillDictionary";

export type GapPriority = "high" | "medium" | "low";

export interface SkillGap {
  skill: string;
  priority: GapPriority;
  evidence: string;
}

export interface MatchResult {
  status: "OK" | "INSUFFICIENT_DATA";
  score: number | null;
  matchedSkills: string[];
  missingSkills: string[];
  skillGaps: SkillGap[];
}

// Interview evidence from earlier mock interviews. It changes what the candidate is credited with.
export interface InterviewEvidence {
  strongSkills: string[];
  weakSkills: Map<string, string>; // skill -> evidence text
}

const PRIORITY_ORDER: Record<GapPriority, number> = { high: 0, medium: 1, low: 2 };

// A skill named three or more times in the posting is core to the role.
const priorityFromMentions = (mentions: number): GapPriority => {
  if (mentions >= 3) return "high";
  if (mentions === 2) return "medium";
  return "low";
};

export const calculateMatch = (
  resumeSkills: string[],
  jobSkills: string[],
  jobDescription?: string,
  interview?: InterviewEvidence,
): MatchResult => {
  const requiredSkills = normalizeSkills(jobSkills);

  if (requiredSkills.length === 0) {
    return {
      status: "INSUFFICIENT_DATA",
      score: null,
      matchedSkills: [],
      missingSkills: [],
      skillGaps: [],
    };
  }

  const weakSkills = interview?.weakSkills ?? new Map<string, string>();

  // Credit comes from the resume and from strong interview evidence.
  // A skill flagged as weak in an interview is not credited, even if the resume lists it.
  const candidateSkills = new Set(normalizeSkills(resumeSkills));
  for (const skill of interview?.strongSkills ?? []) candidateSkills.add(skill);
  for (const skill of weakSkills.keys()) candidateSkills.delete(skill);

  const matchedSkills = requiredSkills.filter((skill) => candidateSkills.has(skill));
  const missingSkills = requiredSkills.filter((skill) => !candidateSkills.has(skill));

  const mentions = jobDescription ? countSkillMentions(jobDescription) : null;

  const skillGaps: SkillGap[] = missingSkills
    .map((skill) => {
      if (weakSkills.has(skill)) {
        return {
          skill,
          priority: "high" as const,
          evidence: "Weak in a mock interview. Practise it before the next interview.",
        };
      }
      const count = mentions?.get(skill) ?? 0;
      const priority: GapPriority = mentions ? priorityFromMentions(count) : "medium";
      const evidence =
        count > 0
          ? `Named ${count} ${count === 1 ? "time" : "times"} in the job description`
          : "Listed in the job requirements";
      return { skill, priority, evidence };
    })
    .sort(
      (a, b) =>
        PRIORITY_ORDER[a.priority] - PRIORITY_ORDER[b.priority] ||
        a.skill.localeCompare(b.skill),
    );

  return {
    status: "OK",
    score: Math.round((matchedSkills.length / requiredSkills.length) * 100),
    matchedSkills,
    missingSkills,
    skillGaps,
  };
};

export const ALIAS_TO_CANONICAL: ReadonlyMap<string, string> = new Map(
  SKILL_DICTIONARY.flatMap((entry) => [
    [entry.canonical, entry.canonical] as const,
    ...entry.aliases.map((alias) => [alias, entry.canonical] as const),
  ]),
);

const escapeRegex = (text: string): string => text.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

// One pattern per skill: all its names as a single alternation, longest first,
// so "Node.js" counts once as node.js and not again as "node".
const SEARCH_PATTERNS: ReadonlyArray<{ canonical: string; pattern: RegExp }> =
  SKILL_DICTIONARY.map((entry) => {
    const names = [...new Set([entry.canonical, ...entry.aliases])].sort(
      (a, b) => b.length - a.length,
    );
    const alternation = names.map(escapeRegex).join("|");
    return {
      canonical: entry.canonical,
      // Whole-word boundaries that still work for names ending in symbols, such as "c++".
      pattern: new RegExp(`(?<![a-z0-9])(?:${alternation})(?![a-z0-9])`, "g"),
    };
  });

// Counts how many times each canonical skill is named in free text.
export const countSkillMentions = (text: string): Map<string, number> => {
  const lower = text.toLowerCase().replace(/[\u2018\u2019]/g, "'");
  const counts = new Map<string, number>();

  for (const { canonical, pattern } of SEARCH_PATTERNS) {
    const hits = lower.match(pattern)?.length ?? 0;
    if (hits > 0) {
      counts.set(canonical, (counts.get(canonical) ?? 0) + hits);
    }
  }
  return counts;
};

// Turns any skill string into its canonical form, e.g. "  ReactJS " -> "react".
// Skills outside the dictionary keep their cleaned lowercase form, so exact matches still work.
import { ALIAS_TO_CANONICAL } from "./skillDictionary";

export const normalizeSkill = (skill: string): string => {
  const cleaned = skill
    .toLowerCase()
    .replace(/\s+/g, " ")
    .replace(/[\s,;:]+$/g, "")
    .trim();

  return ALIAS_TO_CANONICAL.get(cleaned) ?? cleaned;
};

export const normalizeSkills = (skills: string[]): string[] => {
  const unique = new Set<string>();
  for (const skill of skills) {
    const normalized = normalizeSkill(skill);
    if (normalized) unique.add(normalized);
  }
  return [...unique];
};

import { z } from "zod";

export const InterviewRoundSchema = z.object({
  name: z.string().min(1, "Round name is required"),
  duration: z.number().positive("Duration must be a positive number"),
});

export const InterviewPlanSchema = z.object({
  company: z.string().min(1, "Company name is required"),
  role: z.string().min(1, "Role is required"),
  matchScore: z.number().min(0).max(100).nullable(),
  topSkillGaps: z.array(z.string()),
  estimatedDuration: z.number().positive(),
  difficulty: z.enum(["EASY", "MEDIUM", "HARD"]),
  targetSkills: z.array(z.string()),
  rounds: z
    .array(InterviewRoundSchema)
    .min(1, "At least one round is required"),
});

export type InterviewRound = z.infer<typeof InterviewRoundSchema>;
export type InterviewPlan = z.infer<typeof InterviewPlanSchema>;

export const InterviewQuestionSchema = z.object({
  round: z.string().min(1, "Round name cannot be empty"),
  question: z.string().min(1, "Question text cannot be empty"),
  order: z.number().int().positive("Order must be a positive integer"),
});

export const InterviewQuestionsSchema = z.object({
  questions: z
    .array(InterviewQuestionSchema)
    .min(1, "At least one question is required"),
});

export type InterviewQuestion = z.infer<typeof InterviewQuestionSchema>;
export type InterviewQuestions = z.infer<typeof InterviewQuestionsSchema>;

export const QuestionEvaluationSchema = z.object({
  score: z.number().min(0).max(100),
  feedback: z.string().min(1),
  strengths: z.array(z.string()),
  improvements: z.array(z.string()),
});

export type QuestionEvaluation = z.infer<typeof QuestionEvaluationSchema>;
